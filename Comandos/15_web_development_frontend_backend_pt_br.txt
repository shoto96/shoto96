1. BACKEND (NODE.JS / EXPRESS)
const express = require('express');
const app = express();
app.use(express.json());

app.get('/api/health', (req, res) => {
    res.json({ status: 'ok', timestamp: new Date() });
});

app.listen(3000, () => console.log('Server running on port 3000'));
